#### Kamus
    a, loop, hasil, bil1, bil2
#### Algoritma
    For loop <- 0 to inf
        input (a)
        if a > 99 then
            output ("Masukkan bilangan dua digit saja)
        else then
            end for
    bil1 <- a / 10
    bil2 <- a % 10
    hasil <- bil1 * 1000 + bil1 * 100 + bil2 * 10 + bil2
    outout (hasil)
endprogram
